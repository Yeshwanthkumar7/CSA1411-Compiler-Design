
%{
int pos = 0, neg = 0;
%}

%%
-[0-9]+     { neg++; }
[+]?[0-9]+  { pos++; }
[ \t\n]+    ;
.           ;
%%

int main()
{
    printf("Enter numbers separated by spaces:\n");
    yylex();

    printf("Positive numbers: %d\n", pos);
    printf("Negative numbers: %d\n", neg);

    return 0;
}

int yywrap()
{
    return 1;
}
